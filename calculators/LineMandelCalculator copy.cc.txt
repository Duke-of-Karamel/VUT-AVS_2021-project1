int* LineMandelCalculator::calculateMandelbrot()
{
	int* pdata = data;
	float* preal = temp_real;
	float* pimag = temp_imag;

	int i,j,n_it;
	int cancel;

	float zReal, zImag;
	float r2, i2;
	int curr,temp,temp2;

	for (i = 0; i < height; i++)
	{
		// iterate whole rows
		for (n_it = 0; n_it<limit; n_it++)
		{
			cancel = 0;
			#pragma omp simd
			for (j = 0; j < width; j++){
				cancel |= pdata[j];
			}
			if (cancel > 0)
				break;

			#pragma omp simd
			for (j = 0; j < width; j++)
			{
				zReal = preal[j];
				zImag = pimag[j];

				//////////////////////
				r2 = zReal * zReal;
				i2 = zImag * zImag;

				// if (i == height/2 && j == width/2){
				// 	std::cout<< "------------------------\n";
				// 	std::cout<<"real= "<< zReal<< "\timag= "<< zImag<<"\n";
				// 	std::cout<<"imag= 2 * "<< zReal<<" * "<< zImag << " + "<< template_col[i] <<"\n";
				// 	std::cout<<"real= "<< r2 <<" - "<< i2 << " + "<< template_row[j] <<"\n";
				// }

				// if (r2 + i2 > 4.0f)
				// 	return n_it;

				// (A + Bi)^2 = A^2 + 2ABi - B^2
				pimag[j] = 2.0f * zReal * zImag + template_col[i];
				preal[j] = r2 - i2 + template_row[j];

				// if (i == height/2 && j == width/2){
				// 	std::cout<<"imag= "<< temp_imag[j] <<"\n";
				// }
				//////////////////////

				curr = pdata[j];
				temp = (r2 + i2 > 4.0f)? n_it : curr;
				temp2 = (n_it == limit-1)? limit : temp;
				pdata[j] = (curr < 0)? temp2 : curr;

				// if (i == height/2 && j == width/2){
				// 	std::cout << "temp = ("<<r2<<" + "<<i2<<" > 4.0f)? "<<n_it<<" : " << curr << "\n"
				// 				<< "temp2 = ("<<n_it<<" == "<<limit<<"-1)? "<<limit<<" : "<<temp<<"\n"
				// 				<< "pdata[j] = ("<<curr<<" < 0)? "<<temp2<<" : "<<curr<<"\n"
				// 				<< "pdata[j]= "<<pdata[j]<<"\n";
				// }
			}
		}
		preal += width;
		pimag += width;
		pdata += width;
	}
	return data;
}
